```python
def compute_wacc(ticker: str, conn: sqlite3.Connection) -> tuple[float, float, float, bool]:
    """
    Returns (wacc, selic_used, cds_used, used_fallback).
    D-09: Ke = selic + beta*ERP + CDS_brazil; WACC = Ke*(1-D/V) + Kd*(1-t)*(D/V)
    D-10: Falls back to sectors.yaml static values when macro data is stale.
    CDS Brasil stored as decimal (e.g. 0.0155) — do NOT divide by 100.
    """
    cfg = SectorConfig.for_ticker(ticker)
    a = cfg.dcf_assumptions if not cfg.is_bank_model else cfg.gordon_assumptions

    rows = conn.execute("""
        SELECT series_code, value, date FROM macro_series
        WHERE series_code IN (11, 29039)
        ORDER BY date DESC
    """).fetchall()

    macro: dict[int, tuple[float, str]] = {}
    for r in rows:
        code = r["series_code"]
        if code not in macro:  # keep only the latest date per series
            macro[code] = (r["value"], r["date"])

    selic_raw, selic_date = macro.get(11, (None, None))
    cds_raw, cds_date = macro.get(29039, (None, None))

    stale = (
        _is_macro_stale(selic_date) or
        _is_macro_stale(cds_date) or
        selic_raw is None or
        cds_raw is None
    )

    if stale:
        log.warning(f"[{ticker}] macro desatualizado — usando fallback sectors.yaml "
                    f"(selic_date={selic_date}, cds_date={cds_date})")
        selic = float(a.get("risk_free", 0.105))
        cds = float(a.get("country_risk", 0.015))
        used_fallback = True
    else:
        selic = float(selic_raw)   # already decimal per [02-02] decision
        cds = float(cds_raw)       # already decimal per [02-02] decision — do NOT /100
        used_fallback = False

    beta = float(a.get("beta", 1.0))
    erp = float(a.get("erp", 0.055))
    kd = float(a.get("cost_of_debt", 0.115))
    tax = float(a.get("tax_rate", 0.27))
    dv = float(a.get("debt_to_capital", 0.25))

    ke = selic + beta * erp + cds
    wacc = ke * (1.0 - dv) + kd * (1.0 - tax) * dv
    return wacc, selic, cds, used_fallback


def _is_macro_stale(iso_date: str | None) -> bool:
    """True if date is stale (> 1 B3 business day old) or None."""
    if not iso_date:
        return True
    try:
        d = datetime.fromisoformat(iso_date).date()
        return is_stale(d)
    except (ValueError, TypeError):
        return True
```

```python
def _validate_dcf_inputs(
    ticker: str,
    wacc: float,
    terminal_growth: float,
    current_price: float | None,
    price_target: float | None,
) -> str | None:
    """
    Returns confidence_flag string or None if valid.
    FIN-04: Must be called BEFORE run_dcf() — run_dcf() has no internal guard.
    T-DCF-01 mitigation: blocks terminal_growth >= wacc to prevent infinity.
    """
    if wacc <= 0.05:
        log.error(f"[{ticker}] DCF inválido: WACC={wacc:.2%} <= 5% — abortando")
        return "INPUT_INVALIDO"
    if terminal_growth >= wacc:
        log.error(
            f"[{ticker}] DCF inválido: terminal_growth={terminal_growth:.2%} >= "
            f"WACC={wacc:.2%} — divisão por zero no valor terminal"
        )
        return "INPUT_INVALIDO"
    if current_price and current_price > 0 and price_target is not None:
        ratio = price_target / current_price
        if ratio < 0.1 or ratio > 5.0:
            log.warning(f"[{ticker}] fair_value fora do intervalo: {ratio:.2f}x preço atual")
            return "FORA DO INTERVALO CONFIAVEL"
    return None
```

```python
def _compute_dcf_industrial(
    ticker: str,
    conn: sqlite3.Connection,
    ltm: dict,
    cfg: SectorConfig,
    computed_date: str,
) -> None:
    """
    Compute and write financial_dcf row for industrial tickers.
    Handles both dcf_fcff and ev_ebitda_multiple valuation methods.
    Never calls run_dcf() without prior input validation (FIN-04 / T-DCF-01).
    """
    wacc, selic_used, cds_used, used_fallback = compute_wacc(ticker, conn)
    a = cfg.dcf_assumptions
    terminal_growth = float(a.get("terminal_growth", 0.04))
    current_price = _get_current_price(ticker, conn)
    shares = ltm.get("shares_outstanding")

    if cfg.valuation_method == "ev_ebitda_multiple":
        # EV/EBITDA multiple path (PETR4, VALE3, VIVT3, GGBR4, etc.)
        ev_ebitda_a = a.get("ev_ebitda_assumptions", {})
        target_multiple = float(ev_ebitda_a.get("target_multiple_base", 6.0))
        ebitda = ltm.get("ebitda")
        net_debt = ltm.get("net_debt")

        if ebitda is not None and net_debt is not None and shares and shares > 0:
            ev = ebitda * target_multiple
            equity_value = ev - net_debt
            fair_value = equity_value / shares
        else:
            fair_value = None
            log.warning(f"[{ticker}] EV/EBITDA: ebitda/net_debt/shares ausentes — fair_value=NULL")

        upside_pct = ((fair_value - current_price) / current_price
                      if fair_value and current_price and current_price > 0 else None)
        confidence_flag = _validate_dcf_inputs(ticker, wacc, terminal_growth, current_price, fair_value)
        valuation_method_stored = "ev_ebitda"

    else:
        # dcf_fcff path (default industrial)
        # Input validation BEFORE run_dcf() — T-DCF-01 mitigation
        pre_flag = _validate_dcf_inputs(ticker, wacc, terminal_growth, current_price, None)
        if pre_flag == "INPUT_INVALIDO":
            _write_dcf_row(conn, ticker, computed_date, "dcf_fcff", None, None,
                           wacc, terminal_growth, selic_used, cds_used, used_fallback,
                           "INPUT_INVALIDO")
            return

        base_revenue = ltm.get("net_revenue") or 0.0
        base_ebitda = ltm.get("ebitda") or 0.0
        net_debt = ltm.get("net_debt") or 0.0

        try:
            assumptions = DCFAssumptions(**{k: a[k] for k in DCFAssumptions.__dataclass_fields__ if k in a})
            # Override live WACC into assumptions
            assumptions_dict = {**a, "wacc": wacc, "terminal_growth": terminal_growth}
            result = run_dcf(
                ticker=ticker,
                base_year=int(computed_date[:4]),
                base_revenue=base_revenue,
                base_ebitda=base_ebitda,
                assumptions=DCFAssumptions(**{k: assumptions_dict[k]
                                              for k in DCFAssumptions.__dataclass_fields__
                                              if k in assumptions_dict}),
                net_debt=net_debt,
                shares_outstanding=shares or 1.0,
            )
            fair_value = getattr(result, "price_target", None)
        except Exception as exc:
            log.error(f"[{ticker}] run_dcf() falhou: {exc}")
            _write_dcf_row(conn, ticker, computed_date, "dcf_fcff", None, None,
                           wacc, terminal_growth, selic_used, cds_used, used_fallback,
                           "ERRO_CALCULO")
            return

        upside_pct = ((fair_value - current_price) / current_price
                      if fair_value and current_price and current_price > 0 else None)
        confidence_flag = _validate_dcf_inputs(ticker, wacc, terminal_growth, current_price, fair_value)
        valuation_method_stored = "dcf_fcff"

    _write_dcf_row(conn, ticker, computed_date, valuation_method_stored,
                   fair_value, upside_pct, wacc, terminal_growth,
                   selic_used, cds_used, used_fallback, confidence_flag)


def _write_dcf_row(
    conn: sqlite3.Connection, ticker: str, computed_date: str,
    valuation_method: str, fair_value: float | None, upside_pct: float | None,
    wacc: float, terminal_growth: float, selic_used: float, cds_used: float,
    used_fallback: bool, confidence_flag: str | None,
) -> None:
    """Write financial_dcf INSERT OR REPLACE row. Parameterized — never f-string."""
    conn.execute("""
        INSERT OR REPLACE INTO financial_dcf
        (id, ticker, computed_date, valuation_method, fair_value_brl, upside_pct,
         wacc, terminal_growth, selic_used, cds_used, used_fallback, confidence_flag, ingested_at)
        VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
    """, (
        str(uuid.uuid4()), ticker, computed_date, valuation_method,
        fair_value, upside_pct, wacc, terminal_growth,
        selic_used, cds_used, int(used_fallback), confidence_flag,
        datetime.now(timezone.utc).isoformat(),
    ))
    conn.commit()
    log.info(f"[{ticker}] DCF escrito — method={valuation_method} fair_value={fair_value} flag={confidence_flag}")
```
