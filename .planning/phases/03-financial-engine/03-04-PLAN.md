```python
def _compute_bank_model(
    ticker: str,
    conn: sqlite3.Connection,
    ltm: dict,
    cfg: SectorConfig,
    computed_date: str,
) -> None:
    """
    Compute bank DDM fair value and bank multiples.
    FIN-05: Banks use DDM (run_ddm) + bank metrics (calculate_bank_metrics).
    Never uses EBITDA, DCF FCFF, or EV/EBITDA — these are COSIF tickers.
    """
    a = cfg.gordon_assumptions  # bank/holding/utilities use gordon_assumptions, not dcf_assumptions
    current_price = _get_current_price(ticker, conn)
    shares = ltm.get("shares_outstanding")

    # --- DDM fair value ---
    base_net_income = ltm.get("net_income") or 0.0
    payout_ratio = float(a.get("payout_ratio", 0.50))
    terminal_growth = float(a.get("terminal_growth", 0.035))

    # Cost of equity (ke) from WACC formula — compute_wacc() returns (wacc, selic, cds, fallback)
    # For DDM: use ke (not wacc) as cost of equity
    wacc_full, selic_used, cds_used, used_fallback = compute_wacc(ticker, conn)
    # ke = selic + beta*erp + cds (same formula as in compute_wacc; re-derive separately)
    selic = selic_used
    cds = cds_used
    beta = float(a.get("beta", 1.0))
    erp = float(a.get("erp", 0.055))
    ke = selic + beta * erp + cds  # cost of equity for DDM discount

    growth_rates = a.get("base_revenue_growth", [0.07] * 5)

    fair_value = None
    confidence_flag = None

    if base_net_income > 0 and ke > terminal_growth and ke > 0.05:
        try:
            result = run_ddm(
                ticker=ticker,
                base_year=int(computed_date[:4]),
                base_net_income=base_net_income,
                payout_ratio=payout_ratio,
                coe=ke,
                g=terminal_growth,
                growth_rates=growth_rates,
            )
            fair_value = getattr(result, "price_target", None)
            if fair_value and current_price and current_price > 0:
                ratio = fair_value / current_price
                if ratio < 0.1 or ratio > 5.0:
                    confidence_flag = "FORA DO INTERVALO CONFIAVEL"
                    log.warning(f"[{ticker}] DDM: fair_value fora do intervalo: {ratio:.2f}x")
        except Exception as exc:
            log.error(f"[{ticker}] run_ddm() falhou: {exc}")
            confidence_flag = "ERRO_CALCULO"
    elif ke <= terminal_growth:
        log.error(f"[{ticker}] DDM inválido: ke={ke:.2%} <= g={terminal_growth:.2%}")
        confidence_flag = "INPUT_INVALIDO"
    elif ke <= 0.05:
        log.error(f"[{ticker}] DDM inválido: ke={ke:.2%} <= 5%")
        confidence_flag = "INPUT_INVALIDO"

    upside_pct = ((fair_value - current_price) / current_price
                  if fair_value and current_price and current_price > 0 else None)

    _write_dcf_row(
        conn, ticker, computed_date, "ddm",
        fair_value, upside_pct,
        ke,            # store ke (cost of equity) in wacc column for DDM tickers
        terminal_growth, selic_used, cds_used, used_fallback, confidence_flag,
    )
    log.info(f"[{ticker}] DDM escrito — fair_value={fair_value} ke={ke:.3f} flag={confidence_flag}")
```
