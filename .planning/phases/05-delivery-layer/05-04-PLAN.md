Imports:

class ReportGenerator(FPDF):

    def header(self):
        # Called automatically on each add_page()
        # Title stored in self._doc_title (set in generate())
        self.set_font("Helvetica", style="B", size=14)
        self.cell(0, 10, getattr(self, "_doc_title", "Relatório"), align="C", new_x="LMARGIN", new_y="NEXT")
        self.set_draw_color(30, 58, 95)  # Navy #1E3A5F
        self.line(self.l_margin, self.get_y(), self.w - self.r_margin, self.get_y())
        self.ln(4)

    def footer(self):
        self.set_y(-15)
        self.set_font("Helvetica", style="I", size=8)
        self.set_text_color(107, 114, 128)
        self.cell(0, 10, f"Página {self.page_no()}", align="C")
        self.set_text_color(0, 0, 0)

    def _section_title(self, title: str) -> None:
        """Helper: renders a section heading in navy 11pt Bold."""
        self.set_font("Helvetica", style="B", size=11)
        self.set_text_color(30, 58, 95)
        self.cell(0, 8, title, new_x="LMARGIN", new_y="NEXT")
        self.set_text_color(0, 0, 0)
        self.ln(2)

    def _header_section(self, ticker: str, generated_at: str) -> None:
        self.set_font("Helvetica", size=10)
        self.cell(0, 6, f"Ativo: {ticker}", new_x="LMARGIN", new_y="NEXT")
        self.cell(0, 6, f"Gerado em: {generated_at}", new_x="LMARGIN", new_y="NEXT")
        self.ln(4)

    def _thesis_section(self, thesis: dict) -> None:
        self._section_title("Tese de Investimento")
        positioning = thesis.get("positioning", "—")
        # Color the positioning word
        color_map = {
            "COMPRAR": (22, 163, 74),
            "MANTER":  (202, 138, 4),
            "VENDER":  (220, 38, 38),
        }
        r, g, b = color_map.get(positioning, (0, 0, 0))
        self.set_font("Helvetica", style="B", size=11)
        self.set_text_color(r, g, b)
        self.cell(0, 7, f"Posicionamento: {positioning}", new_x="LMARGIN", new_y="NEXT")
        self.set_text_color(0, 0, 0)
        self.set_font("Helvetica", size=10)
        self.cell(0, 6, f"Confiança: {thesis.get('confidence', '—')}", new_x="LMARGIN", new_y="NEXT")
        self.ln(2)
        self.set_font("Helvetica", size=10)
        self.multi_cell(0, 5, thesis.get("rationale", "—"))
        self.ln(2)
        self.set_font("Helvetica", style="I", size=9)
        self.multi_cell(0, 5, thesis.get("summary_one_line", ""))
        self.ln(4)

    def _bull_bear_section(self, thesis: dict) -> None:
        self._section_title("Cenário Otimista")
        self.set_font("Helvetica", size=10)
        self.multi_cell(0, 5, thesis.get("bull_case", "—"))
        self.ln(4)
        self._section_title("Cenário Pessimista")
        self.set_font("Helvetica", size=10)
        self.multi_cell(0, 5, thesis.get("bear_case", "—"))
        self.ln(4)

    def _drivers_risks_section(self, drivers: list, risks: list) -> None:
        # Drivers table
        self._section_title("Drivers de Investimento")
        self.set_font("Helvetica", size=9)
        with self.table(col_widths=(55, 115, 20), borders_layout="SINGLE_TOP_LINE") as table:
            hrow = table.row()
            for h in ("Driver", "Descrição", "Impacto"):
                hrow.cell(h)
            for d in drivers:
                row = table.row()
                row.cell(str(d.get("title", "—"))[:50])
                row.cell(str(d.get("description", "—"))[:200])
                row.cell(str(d.get("impact", "—")))
        self.ln(4)

        # Risks table
        self._section_title("Riscos")
        self.set_font("Helvetica", size=9)
        with self.table(col_widths=(55, 115, 20), borders_layout="SINGLE_TOP_LINE") as table:
            hrow = table.row()
            for h in ("Risco", "Descrição", "Severidade"):
                hrow.cell(h)
            for r in risks:
                row = table.row()
                row.cell(str(r.get("title", "—"))[:50])
                row.cell(str(r.get("description", "—"))[:200])
                row.cell(str(r.get("severity", "—")))
        self.ln(4)

    def _valuation_section(self, dcf: dict, multiples: dict) -> None:
        self._section_title("Valuation")
        self.set_font("Helvetica", size=10)
        fv   = dcf.get("fair_value_brl")
        up   = dcf.get("upside_pct")
        wacc = dcf.get("wacc")
        tg   = dcf.get("terminal_growth")
        pe   = multiples.get("pe_ratio")
        ev   = multiples.get("ev_ebitda")
        pbv  = multiples.get("pbv_ratio")
        dy   = multiples.get("dividend_yield")
        px   = multiples.get("price")

        def _fmt_float(v, decimals=2, prefix="", suffix=""):
            if v is None:
                return "—"
            return f"{prefix}{v:.{decimals}f}{suffix}"

        rows_data = [
            ("Valor Justo (R$)",        _fmt_float(fv, 2, "R$ ")),
            ("Upside %",                _fmt_float(up, 1, suffix="%")),
            ("WACC",                    _fmt_float(wacc * 100 if wacc else None, 2, suffix="%")),
            ("Crescimento Terminal",    _fmt_float(tg * 100 if tg else None, 2, suffix="%")),
            ("Preço Atual (R$)",        _fmt_float(px, 2, "R$ ")),
            ("P/L",                     _fmt_float(pe, 1, suffix="x")),
            ("EV/EBITDA",               _fmt_float(ev, 1, suffix="x")),
            ("P/VPA",                   _fmt_float(pbv, 2, suffix="x")),
            ("Dividend Yield",          _fmt_float(dy * 100 if dy else None, 2, suffix="%")),
        ]
        self.set_font("Helvetica", size=9)
        with self.table(col_widths=(90, 100), borders_layout="SINGLE_TOP_LINE") as table:
            hrow = table.row()
            hrow.cell("Indicador")
            hrow.cell("Valor")
            for label, value in rows_data:
                row = table.row()
                row.cell(label)
                row.cell(value)
        self.ln(4)

    def _financials_section(self, ltm: dict) -> None:
        self._section_title("Destaques Financeiros (LTM)")
        self.set_font("Helvetica", size=9)
        def _millions(v):
            if v is None:
                return "—"
            return f"R$ {v / 1_000_000:.1f}M"
        rows_data = [
            ("Receita Líquida",   _millions(ltm.get("net_revenue"))),
            ("EBITDA",            _millions(ltm.get("ebitda"))),
            ("Lucro Líquido",     _millions(ltm.get("net_income"))),
            ("FCF",               _millions(ltm.get("fcf"))),
            ("Dívida Líquida",    _millions(ltm.get("net_debt"))),
        ]
        with self.table(col_widths=(90, 100), borders_layout="SINGLE_TOP_LINE") as table:
            hrow = table.row()
            hrow.cell("Métrica")
            hrow.cell("Valor (12 meses)")
            for label, value in rows_data:
                row = table.row()
                row.cell(label)
                row.cell(value)
        self.ln(4)

    def _macro_section(self, macro: dict) -> None:
        """macro dict: {selic:[{date,value},...], ipca_12m:[...], ptax:[...], cds_brasil:[...]}"""
        self._section_title("Contexto Macroeconômico")
        self.set_font("Helvetica", size=9)

        def _latest(series_list: list) -> str:
            if not series_list:
                return "—"
            latest = series_list[-1]  # already ordered chronologically by get_macro_panel()
            v = latest.get("value")
            return f"{v:.4f}" if v is not None else "—"

        rows_data = [
            ("Selic (% a.a.)",    _latest(macro.get("selic", []))),
            ("IPCA Acum. 12m (%)", _latest(macro.get("ipca_12m", []))),
            ("PTAX (R$/USD)",     _latest(macro.get("ptax", []))),
            ("CDS Brasil (bps)",  _latest(macro.get("cds_brasil", []))),
        ]
        with self.table(col_widths=(100, 90), borders_layout="SINGLE_TOP_LINE") as table:
            hrow = table.row()
            hrow.cell("Série")
            hrow.cell("Último Valor")
            for label, value in rows_data:
                row = table.row()
                row.cell(label)
                row.cell(value)
        self.ln(6)

    def _disclaimer(self) -> None:
        self.set_font("Helvetica", style="I", size=8)
        self.set_text_color(107, 114, 128)  # grey per UI-SPEC.md
        self._section_title("Aviso Legal")
        self.set_font("Helvetica", style="I", size=8)
        self.set_text_color(107, 114, 128)
        self.multi_cell(0, 4, _CVM_DISCLAIMER)
        self.set_text_color(0, 0, 0)

    def generate(self, ticker: str, data: dict) -> bytes:
        """Generate PDF report for ticker using provided data dict. Returns bytes.
        Per D-17: in-memory only — no file writes.
        """
        # Validate caller is using a fresh instance (Pitfall 4 guard)
        thesis   = data.get("thesis", {})
        dcf      = data.get("dcf") or {}
        ltm      = data.get("ltm") or {}
        multiples = data.get("multiples") or {}
        macro    = data.get("macro") or {}  # may be empty in fixture; macro_section handles {}
        generated_at = data.get("generated_at", "—")

        self._doc_title = f"Relatório de Investimento — {ticker}"
        self.set_margins(left=20, top=25, right=20)
        self.set_auto_page_break(auto=True, margin=20)
        self.add_page()

        self._header_section(ticker, generated_at)
        self._thesis_section(thesis)
        self._bull_bear_section(thesis)
        self._drivers_risks_section(thesis.get("drivers", []), thesis.get("risks", []))
        self._valuation_section(dcf, multiples)
        self._financials_section(ltm)
        self._macro_section(macro)
        self._disclaimer()  # always last

        return bytes(self.output())  # in-memory — no dest='S' (old fpdf API)

    def generate_from_fixture(self, ticker: str) -> bytes:
        """Generate PDF using hardcoded fixture data. Used for tests only.
        Returns fresh ReportGenerator instance result to avoid Pitfall 4.
        """
        # Create a new instance to avoid reuse-after-output() issue
        fresh = ReportGenerator()
        return fresh.generate(ticker, _FIXTURE)
